cmake_minimum_required(VERSION 3.16)
project(IndustryAlgorithm LANGUAGES CXX)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT fingerprint_compare)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/DNOMINMAX>)
endif()

option(FINGERPRINT_DEEPPRINT "Build the DeepPrint KISA DLL alongside the evaluation tools" OFF)
option(FINGERPRINT_FVC2004_ROI "Crop foreground in DeepPrint KISA iType 3 inputs" ON)
set(FINGERPRINT_TEXTURE_WEIGHT "0.22" CACHE STRING "Texture branch weight for DeepPrint matching")
option(FINGERPRINT_BUILD_GUI "Build the Windows KISA evaluation application" ON)
option(FINGERPRINT_BUILD_RESEARCH "Build native DeepPrint research tools" OFF)
set(FINGERPRINT_TORCH_NATIVE_DIR "" CACHE PATH "TorchSharp 0.105.0 and LibTorch CPU 2.5.1 native DLL directory for C++ training")
set(FINGERPRINT_ONNXRUNTIME_ROOT "${CMAKE_SOURCE_DIR}/build/dependencies/onnxruntime" CACHE PATH "Extracted Windows x64 ONNX Runtime NuGet package")
set(FINGERPRINT_DEEPPRINT_MODEL "${CMAKE_SOURCE_DIR}/build/models/deepprint_texminu_512.onnx" CACHE FILEPATH "Exported DeepPrint ONNX model")
set(FINGERPRINT_DEEPPRINT_REFERENCE_DIR "" CACHE PATH "Reference vectors from the DeepPrint exporter")
get_filename_component(FINGERPRINT_ONNXRUNTIME_ROOT "${FINGERPRINT_ONNXRUNTIME_ROOT}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
get_filename_component(FINGERPRINT_DEEPPRINT_MODEL "${FINGERPRINT_DEEPPRINT_MODEL}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
if(FINGERPRINT_DEEPPRINT_REFERENCE_DIR)
    get_filename_component(FINGERPRINT_DEEPPRINT_REFERENCE_DIR "${FINGERPRINT_DEEPPRINT_REFERENCE_DIR}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
endif()

if(FINGERPRINT_DEEPPRINT)
    enable_language(RC)
    add_library(IndustryAlgorithm SHARED
        src/api/kisa_api.cpp
        src/feature/deepprint_extractor.cpp
        src/feature/deepprint_preprocessing.cpp
        src/matching/fingerprint_matcher.cpp
        src/feature/template_manager.cpp
    )
    target_include_directories(IndustryAlgorithm PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" PRIVATE src)
    target_compile_definitions(IndustryAlgorithm PRIVATE
        FINGERPRINT_TEXTURE_WEIGHT=${FINGERPRINT_TEXTURE_WEIGHT}
        FINGERPRINT_FVC2004_ROI=$<BOOL:${FINGERPRINT_FVC2004_ROI}>)
    set_target_properties(IndustryAlgorithm PROPERTIES PREFIX "")
    if(MINGW)
        target_link_options(IndustryAlgorithm PRIVATE -static -static-libstdc++)
    endif()

    set(ort_include "${FINGERPRINT_ONNXRUNTIME_ROOT}/build/native/include")
    set(ort_runtime "${FINGERPRINT_ONNXRUNTIME_ROOT}/runtimes/win-x64/native")
    set(ort_dll "${ort_runtime}/onnxruntime.dll")
    if(NOT FINGERPRINT_ONNXRUNTIME_ROOT OR NOT EXISTS "${ort_include}/onnxruntime_cxx_api.h"
       OR NOT EXISTS "${ort_dll}" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "DeepPrint requires a Windows x64 build and FINGERPRINT_ONNXRUNTIME_ROOT pointing to the extracted Microsoft.ML.OnnxRuntime NuGet package")
    endif()
    if(NOT EXISTS "${FINGERPRINT_DEEPPRINT_MODEL}")
        message(FATAL_ERROR "Set FINGERPRINT_DEEPPRINT_MODEL to the ONNX file generated by export_deepprint")
    endif()

    # Regenerate the RC source when the ONNX changes, including for Visual Studio builds.
    set(model_rc "${CMAKE_CURRENT_BINARY_DIR}/deepprint_model.rc")
    configure_file(src/feature/deepprint_model.rc.in "${model_rc}.in" @ONLY)
    add_custom_command(OUTPUT "${model_rc}"
        COMMAND "${CMAKE_COMMAND}" -E copy "${model_rc}.in" "${model_rc}"
        DEPENDS "${model_rc}.in" "${FINGERPRINT_DEEPPRINT_MODEL}"
        VERBATIM)
    target_sources(IndustryAlgorithm PRIVATE "${model_rc}")
    target_include_directories(IndustryAlgorithm SYSTEM PRIVATE "${ort_include}")
    set_property(TARGET IndustryAlgorithm APPEND PROPERTY LINK_DEPENDS "${ort_dll}")
    add_custom_command(TARGET IndustryAlgorithm POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ort_dll}" "$<TARGET_FILE_DIR:IndustryAlgorithm>"
        VERBATIM
    )
    install(FILES "${ort_dll}" DESTINATION . COMPONENT KisaAlgorithm)
    install(TARGETS IndustryAlgorithm RUNTIME DESTINATION . COMPONENT KisaAlgorithm)
endif()

include(CTest)
if(FINGERPRINT_BUILD_RESEARCH)
    add_library(deepprint_onnx STATIC tools/deepprint/onnx_model.cpp)
    target_compile_definitions(deepprint_onnx PRIVATE NOMINMAX)
    target_link_libraries(deepprint_onnx PUBLIC bcrypt)
    add_executable(export_deepprint tools/deepprint/export_model.cpp)
    target_link_libraries(export_deepprint PRIVATE deepprint_onnx)
    set(research_ort "${FINGERPRINT_ONNXRUNTIME_ROOT}/runtimes/win-x64/native")
    add_library(deepprint_studies STATIC tools/deepprint/research_data.cpp
        tools/deepprint/cache.cpp tools/deepprint/roi_study.cpp tools/deepprint/preprocessing_study.cpp
        src/feature/deepprint_preprocessing.cpp src/feature/template_manager.cpp
        src/matching/fingerprint_matcher.cpp)
    target_include_directories(deepprint_studies PUBLIC include src tools/deepprint)
    target_include_directories(deepprint_studies SYSTEM PRIVATE "${FINGERPRINT_ONNXRUNTIME_ROOT}/build/native/include")
    target_link_libraries(deepprint_studies PUBLIC deepprint_onnx fingerprint_tiff verification_metrics)
    if(MINGW)
        target_link_libraries(deepprint_studies PRIVATE "${research_ort}/onnxruntime.dll")
    else()
        target_link_libraries(deepprint_studies PRIVATE "${research_ort}/onnxruntime.lib")
    endif()
    add_executable(deepprint_research tools/deepprint/research_main.cpp)
    target_link_libraries(deepprint_research PRIVATE deepprint_studies)
    add_custom_command(TARGET deepprint_research POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${research_ort}/onnxruntime.dll" "$<TARGET_FILE_DIR:deepprint_research>" VERBATIM)
    if(BUILD_TESTING)
        add_executable(test_research tests/test_research.cpp)
        target_link_libraries(test_research PRIVATE deepprint_studies)
        if(MINGW)
            target_link_options(test_research PRIVATE -static)
        endif()
        add_custom_command(TARGET test_research POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${research_ort}/onnxruntime.dll" "$<TARGET_FILE_DIR:test_research>" VERBATIM)
        add_test(NAME research_metrics_and_onnx COMMAND test_research)
    endif()
    if(MINGW)
        target_link_options(export_deepprint PRIVATE -municode -static)
        target_link_options(deepprint_research PRIVATE -municode -static)
    endif()
    if(FINGERPRINT_TORCH_NATIVE_DIR)
        get_filename_component(FINGERPRINT_TORCH_NATIVE_DIR "${FINGERPRINT_TORCH_NATIVE_DIR}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
        set(torch_bridge "${FINGERPRINT_TORCH_NATIVE_DIR}/LibTorchSharp.dll")
        if(NOT EXISTS "${torch_bridge}" OR NOT EXISTS "${FINGERPRINT_TORCH_NATIVE_DIR}/torch_cpu.dll")
            message(FATAL_ERROR "FINGERPRINT_TORCH_NATIVE_DIR must contain the native training DLLs")
        endif()
        add_executable(deepprint_train tools/deepprint/train_main.cpp tools/deepprint/training.cpp
            tools/deepprint/tail_study.cpp tools/deepprint/legacy_study.cpp)
        target_link_libraries(deepprint_train PRIVATE deepprint_studies)
        if(MINGW)
            target_link_libraries(deepprint_train PRIVATE "${torch_bridge}")
            target_link_options(deepprint_train PRIVATE -municode -static)
        else()
            # NuGet ships the C ABI DLL without an MSVC import library.
            file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tools/deepprint/torch_tensor.h" torch_header)
            string(REGEX MATCHALL "THS(Tensor|NN|Torch)_[A-Za-z0-9_]+" torch_symbols "${torch_header}")
            list(REMOVE_DUPLICATES torch_symbols)
            set(torch_def "LIBRARY LibTorchSharp.dll\nEXPORTS\n")
            foreach(symbol IN LISTS torch_symbols)
                string(APPEND torch_def "${symbol}\n")
            endforeach()
            file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.def" "${torch_def}")
            add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.lib"
                COMMAND "${CMAKE_AR}" /nologo /machine:x64 "/def:${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.def"
                    "/out:${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.lib" VERBATIM)
            target_sources(deepprint_train PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.lib")
            target_link_libraries(deepprint_train PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/torch_bridge.lib")
        endif()
        set(training_dlls "${research_ort}/onnxruntime.dll" "${torch_bridge}")
        foreach(runtime asmjit c10 fbgemm libiomp5md libiompstubs5md torch torch_cpu torch_global_deps uv)
            list(APPEND training_dlls "${FINGERPRINT_TORCH_NATIVE_DIR}/${runtime}.dll")
        endforeach()
        add_custom_command(TARGET deepprint_train POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${training_dlls} "$<TARGET_FILE_DIR:deepprint_train>" VERBATIM)
    endif()
endif()
